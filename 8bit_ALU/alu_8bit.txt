module alu_8bit(
input [7:0] A,
input [7:0] B,
input [2:0] sel,
output reg [7:0] Y,
output reg carry

);

always @(*) begin 
     carry = 1'b0;

     case(sel)
        3'b000: {carry,Y} = A + B;
        3'b001: {carry,Y} = A - B;
        3'b010: Y = ~A ;
        3'b011: Y = A & B;
        3'b100: Y = A | B;
        3'b101: Y = A ^ B;
        3'b110: {carry,Y} = A + 8'b1;
        3'b111: {carry,Y} = A - 8'b1;
        
        default: Y= 8'b0;
     endcase
end

endmodule
