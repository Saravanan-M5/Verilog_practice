module or_gate(output y, input a, input b);
assign y=a|b;
endmodule 

module or_tb;
reg a,b;
wire y;

or_gate dut(.a(a),.b(b),.y(y));
initial begin
$monitor("a=%b b=%b y=%b",a,b,y);
$dumpfile("or_tb.vcd");
$dumpvars(0,or_tb);
a=0; b=0;#25;
a=0; b=1;#25;
a=1; b=0;#25;
a=1; b=1;#25;
$finish;
end
endmodule
