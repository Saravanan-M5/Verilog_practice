module half_add(output sum,output carry, input a, input b);
assign sum=a^b;
assign carry=a&b;

endmodule

module half_add_tb;
reg a,b;
wire sum,carry;

half_add dut(.a(a),.b(b),.sum(sum),.carry(carry));

initial begin 
$monitor("a=%b b=%b | sum=%b carry=%b",a,b,sum,carry);
$dumpfile("half_add_tb.vcd");
$dumpvars(0,half_add_tb);
a=0; b=0; #10;
a=0; b=1; #10;
a=1; b=0; #10;
a=1; b=1; #10;
$finish;
end
endmodule
